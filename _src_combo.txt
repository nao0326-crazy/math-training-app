  38 | /** 順列を数える nPr = n! / (n-r)! (小6なので小さい値のみ) */
  39 | function permutation(n: number, r: number): number {
  40 |   let result = 1;
  41 |   for (let i = 0; i < r; i++) {
  42 |     result *= n - i;
  43 |   }
  44 |   return result;
  45 | }
  46 | 
  47 | /** 組合せの数え上げ nCr */
  48 | function combination(n: number, r: number): number {
  49 |   if (r === 0) return 1;
  50 |   if (r > n) return 0;
  51 |   return permutation(n, r) / factorial(r);
  52 | }
  53 | 
  54 | /** 階乗 */
  55 | function factorial(n: number): number {
  56 |   let result = 1;
  57 |   for (let i = 2; i <= n; i++) result *= i;
  58 |   return result;
  59 | }
  60 | /**
  61 |  * 並べ方の問題の種類 (「n個から一部を選んで並べる」を対象にしたもの)
  62 |  *
  63 |  * arrange_tree (「すべてを並べる」) と明確に区別するため、ここでは必ず r < n
  64 |  * (一部しか選ばない) ことを前提とする。種類ごとに「選ぶ段階」と「並べる段階」の
  65 |  * どちらに条件があるかで、考え方も解法も異なる。
  66 |  *
  67 |  * 採用した種類はいずれも小6の基本的な方法 (かけ算・順序を考えて選ぶ) で解ける。
  68 |  * 組合せの公式 nCr を使う解法 (n! を r! で割る方法) は採用しない。

 100 | /** from 個から count 個を選んで並べる通り数 (かけ算の積で求める) */
 101 | function prod(from: number, count: number): number {
 102 |   let v = 1;
 103 |   for (let i = 0; i < count; i++) v *= from - i;
 104 |   return v;
 105 | }
 106 | 
 107 | /**
 108 |  * 種類ごとの答えを求める。
 109 |  *
 110 |  * pick_special だけは「並べない」問題になるため combination を使う。
 111 |  * それ以外はすべてかけ算の積 (prod) で求められる。
 112 |  */
 113 | function solveArrangeSimple(variant: ArrangeSimpleVariant, n: number, r: number): number {
 114 |   switch (variant) {
 115 |     case 'pick_only':
 116 |       // n個から r個を選んで並べる = n × (n-1) × … の r 個
 117 |       return prod(n, r);
 118 |     case 'pick_special':
 119 |       // 特別の1人を n 人の中から決める n 通り × 残り n-1 人から r-1 人を選ぶ
 120 |       return n * combination(n - 1, r - 1);
 121 |     case 'pick_include_one':
 122 |       // 必ずAさんを含めるので、残り n-1 人から r-1 人を選んで並べる
 123 |       return prod(n - 1, r - 1);
 124 |     case 'pick_first_fixed':
 125 |       // 先頭はAさんなので、残り n-1 人から r-1 人を選んで並べる
 126 |       return prod(n - 1, r - 1);
 127 |     case 'pick_both_ends':
 128 |       // 両端はBさんとCさん (左右2通り)、残り n-2 人から r-2 人を選んで並べる
 129 |       return 2 * prod(n - 2, r - 2);
 130 |   }
 131 | }

 133 | /** 難易度・乱数から、成立する (variant, n, r) を選び出す */
 134 | function pickVariantAndParams(
 135 |   rng: ReturnType<typeof createRandom>,
 136 |   lv: DifficultyLevel,
 137 | ): { variant: ArrangeSimpleVariant; n: number; r: number } {
 138 |   const [minN, maxN] = ARRANGE_SIMPLE_N[lv];
 139 |   const usable = ARRANGE_SIMPLE_VARIANTS[lv];
 140 | 
 141 |   for (let attempt = 0; attempt < 50; attempt++) {
 142 |     const variant = rng.pick(usable);
 143 |     const n = rng.int(minN, maxN);
 144 |     // pick_both_ends は両端に2人固定するので最低3人選ぶ必要がある
 145 |     const minR = variant === 'pick_both_ends' ? 3 : 2;
 146 |     // r < n を保証する (r = n は arrange_tree と重複するため使わない)
 147 |     const maxR = n - 1;
 148 |     if (maxR < minR) continue;
 149 |     return { variant, n, r: rng.int(minR, maxR) };
 150 |   }
 151 | 
 152 |   // フォールバック (必ず成立する最小の組合せ)
 153 |   const fallbackN = Math.max(minN, 4);
 154 |   const fallback =
 155 |     usable.find((v) => (v === 'pick_both_ends' ? fallbackN >= 4 : true)) ?? 'pick_only';
 156 |   return { variant: fallback, n: fallbackN, r: 2 };
 157 | }

 159 | /** 積の式を「n×n-1×…×n-r+1」の文字列にする (count<=1 は1通り) */
 160 | function fallingString(from: number, count: number): string {
 161 |   if (count <= 1) return '1';
 162 |   const parts: string[] = [];
 163 |   for (let i = 0; i < count; i++) parts.push(String(from - i));
 164 |   return parts.join('×');
 165 | }
